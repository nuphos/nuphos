Nuphos backend: {{ include "nuphos.backendUrl" . }}

     kubectl -n {{ .Release.Namespace }} rollout status deploy/{{ include "nuphos.fullname" . }}-backend

2. In Nuphos Desktop, choose "Self-hosted? Set API endpoint", enter the URL
   above, sign in with the emailed code and create a team.
{{- if .Values.runtime.enabled }}

3. Print the runtime console password:
     kubectl -n {{ .Release.Namespace }} get secret {{ include "nuphos.secretName" . }} -o jsonpath='{.data.RUNTIME_AUTH_KEY}' | base64 -d

4. Open {{ include "nuphos.runtimeUrl" . }}, sign in with that password, set the
   public URL to {{ include "nuphos.runtimeUrl" . | replace "https://" "wss://" | replace "http://" "ws://" }}/acp,
   pair it with your team using the single-use code, then sign in to Claude.
{{- end }}
